cmake_minimum_required(VERSION 3.24)
project(HandmadeBTQuick VERSION 0.10.7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2 Widgets Network)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(HandmadeBTQuick WIN32 main.cpp updater.cpp updater.hpp icon.rc)
target_compile_definitions(HandmadeBTQuick PRIVATE EEGO_VERSION="${PROJECT_VERSION}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/release-version.txt" "${PROJECT_VERSION}")
qt_add_qml_module(HandmadeBTQuick URI HandmadeBT VERSION 1.0 QML_FILES Main.qml
    RESOURCES assets/eego-nitro.png assets/eego-face.png assets/eego.ico assets/author-payment.jpg
    assets/inbox.svg assets/download.svg assets/check.svg assets/pause.svg assets/alert.svg assets/settings.svg assets/file.svg assets/plus.svg assets/close.svg assets/search.svg assets/play.svg assets/retry.svg)
target_link_libraries(HandmadeBTQuick PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Network)
option(EEGO_BUILD_SERVICE_TEST "Build isolated frontend service tests" OFF)
if(EEGO_BUILD_SERVICE_TEST)
 qt_add_executable(eego_updater_test updater_test.cpp updater.cpp updater.hpp)
 target_link_libraries(eego_updater_test PRIVATE Qt6::Widgets Qt6::Network)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 qt_add_executable(eego_task_ui_test main.cpp)
 target_compile_definitions(eego_task_ui_test PRIVATE EEGO_SERVICE_TEST EEGO_TASK_UI_TEST)
 target_link_libraries(eego_task_ui_test PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Test)
 qt_add_executable(eego_service_test main.cpp)
 target_compile_definitions(eego_service_test PRIVATE EEGO_SERVICE_TEST)
 target_link_libraries(eego_service_test PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets)
endif()

if(EEGO_BUILD_SERVICE_TEST)
 qt_add_executable(eego_unified_service_test main.cpp)
 target_compile_definitions(eego_unified_service_test PRIVATE EEGO_SERVICE_TEST EEGO_UNIFIED_SERVICE_TEST)
 target_link_libraries(eego_unified_service_test PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets)
 qt_add_executable(eego_continuity_test continuity_test.cpp)
 target_link_libraries(eego_continuity_test PRIVATE Qt6::Core)
endif()
