if(HBT_EEGO_CORE)
  configure_file(apps/Eego/core/secure-dns-hosts.txt secure-dns-hosts.txt COPYONLY)
  # Quoted includes must resolve to the same version in every translation unit.
  set(core_dir "${CMAKE_CURRENT_BINARY_DIR}/eego-src")
  file(GLOB shared_sources CONFIGURE_DEPENDS "src/*.cpp" "src/*.hpp")
  foreach(shared IN LISTS shared_sources)
    get_filename_component(name "${shared}" NAME)
    set(override "${CMAKE_CURRENT_SOURCE_DIR}/apps/Eego/core/${name}")
    if(EXISTS "${override}")
      configure_file("${override}" "${core_dir}/${name}" COPYONLY)
    else()
      configure_file("${shared}" "${core_dir}/${name}" COPYONLY)
    endif()
  endforeach()
  configure_file(apps/Eego/core/qt_library.hpp "${core_dir}/qt_library.hpp" COPYONLY)
  configure_file(apps/Eego/core/bt_peer_budget.hpp "${core_dir}/bt_peer_budget.hpp" COPYONLY)
  configure_file(apps/Eego/core/output_layout.hpp "${core_dir}/output_layout.hpp" COPYONLY)
endif()
